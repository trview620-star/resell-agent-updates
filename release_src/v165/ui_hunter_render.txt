def _set_hunter_view(name):
    st.session_state['hunter_active_view']=name


def render_hunter():
    hunter.initialize()
    market_intelligence.initialize()
    st.markdown("""
       <div class="resell-hero">
         <div class="eyebrow">MARKET INTELLIGENCE &nbsp; / &nbsp; PRODUCT DISCOVERY</div>
         <h1>Találd meg, amit tényleg keresnek.</h1>
         <p>50 adatforrásokra épülő kutatási jelölt, ellenőrizhető magyar
         beszerzési ajánlatok, saját kulcsszavas keresés és természetes hirdetések.
         A keresletet, az ajánlati árakat és a tényleges eladásokat külön kezeljük.</p>
       </div>""",unsafe_allow_html=True)
    serper,cloud_id,cloud_token=credentials()
    markets=st.session_state.get('hunter_marketplaces',list(db.PLATFORMS))
    use_ai=st.session_state.get('hunter_use_ai',True)

    with st.expander('Piacterek és szövegbeállítások',expanded=False):
        st.multiselect('Piacterek',list(db.PLATFORMS),
                       default=list(db.PLATFORMS),key='hunter_marketplaces')
        st.checkbox('Természetes magyar AI-leírás Cloudflare-rel (ha van kulcs)',
                    value=True,key='hunter_use_ai')
        markets=st.session_state.get('hunter_marketplaces',list(db.PLATFORMS))
        use_ai=st.session_state.get('hunter_use_ai',True)

    available=['Top 50','Saját keresés','Kiválasztottak','Közzététel','Beállítások']
    if st.session_state.get('hunter_active_view') not in available:
        st.session_state['hunter_active_view']='Top 50'
    view=st.radio('Munkaterület',available,key='hunter_active_view',horizontal=True)

    if view=='Top 50':
        ui_market_radar.render_top50(serper,cloud_id,cloud_token,markets,use_ai)
    elif view=='Saját keresés':
        ui_market_radar.render_custom(serper,cloud_id,cloud_token,markets,use_ai)
    elif view=='Kiválasztottak':
        _selected_tab()
    elif view=='Közzététel':
        _publishing_tab()
    else:
        _settings_tab()

    st.markdown('<div class="resell-bottomcap">GYORSMŰVELETEK &nbsp;'
                '· &nbsp; A hirdetések piszkozatként készülnek. '
                'Vásárlás és közzététel külön jóváhagyást igényel.</div>',
                unsafe_allow_html=True)
    a,b,c,d=st.columns(4,gap='small')
    for col,label in [
        (a,'Top 50'),(b,'Saját keresés'),
        (c,'Kiválasztottak'),(d,'Közzététel')]:
        col.button(label,key='hunter_bottom_'+label,
                   on_click=_set_hunter_view,args=(label,),
                   use_container_width=True)
